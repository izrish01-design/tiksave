// 1. Pasang pakej: npm init -y && npm install express cors axios fluent-ffmpeg ffmpeg-static
// 2. Jalankan pelayan: node server.js

const express = require('express');
const cors = require('cors');
const axios = require('axios');
const ffmpeg = require('fluent-ffmpeg');
const ffmpegPath = require('ffmpeg-static');
ffmpeg.setFfmpegPath(ffmpegPath);

const app = express();
app.use(cors());
app.use(express.json());

// Endpoint 1: Dapatkan pautan video TikTok tanpa watermark
app.get('/api/tiktok', async (req, res) => {
    const videoUrl = req.query.url;
    if (!videoUrl) return res.status(400).json({ error: 'Parameter URL TikTok diperlukan.' });

    try {
        const response = await axios.get('https://www.tikwm.com/api/', {
            params: { url: videoUrl, hd: 1 },
            headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' }
        });
        return res.json(response.data);
    } catch (err) {
        return res.status(500).json({ error: 'Gagal menghubungi pelayan sumber.', detail: err.message });
    }
});

// Endpoint 2: FFmpeg Upscale/Downscale & Penajaman AI mengikut parameter pengguna
// Endpoint 2: FFmpeg Upscale/Downscale & Penajaman AI tanpa mampatan
app.get('/api/ai-upscale', async (req, res) => {
    const {
        videoUrl,
        quality = '4k_ai',
        ai = '1',
        sharpness = 75,
        contrast = 18,
        saturation = 20,
        brightness = 4,
        denoise = 35
    } = req.query;

    if (!videoUrl) return res.status(400).send('URL video diperlukan.');

    res.setHeader('Content-Disposition', `attachment; filename="TikSave_${quality}.mp4"`);
    res.setHeader('Content-Type', 'video/mp4');

    const scaleMap = {
        '4k_ai': 'scale=2160:-2:flags=lanczos',
        '1080p': 'scale=1080:-2:flags=lanczos',
        '720p': 'scale=720:-2:flags=bicubic',
        '480p': 'scale=480:-2:flags=bilinear'
    };

    const filters = [scaleMap[quality] || scaleMap['1080p']];

    if (ai === '1') {
        const sharpAmount = ((Number(sharpness) / 100) * 1.8).toFixed(2);
        const contrastVal = (1 + Number(contrast) / 100).toFixed(2);
        const satVal = (1 + Number(saturation) / 100).toFixed(2);
        const brightVal = (Number(brightness) / 100).toFixed(2);
        const denoiseVal = ((Number(denoise) / 100) * 6).toFixed(1);

        if (Number(denoise) > 0) filters.push(`hqdn3d=${denoiseVal}:${denoiseVal}:4:4`);
        if (Number(sharpness) > 0) filters.push(`unsharp=5:5:${sharpAmount}:5:5:0.0`);
        filters.push(`eq=contrast=${contrastVal}:saturation=${satVal}:brightness=${brightVal}`);
    }

    ffmpeg(videoUrl)
        .videoFilters(filters.join(','))
        .videoCodec('libx264')
        .outputOptions(['-preset slow', '-crf 16', '-pix_fmt yuv420p', '-movflags frag_keyframe+empty_moov'])
        .format('mp4')
        .on('error', (err) => console.error('Ralat FFmpeg:', err.message))
        .pipe(res, { end: true });
});

app.listen(3000, () => console.log('TikSave AI Backend aktif di http://localhost:3000'));
// Integrasi AI Sebenar menggunakan Model Real-ESRGAN Video Upscaler (Replicate API)
// Pasang: npm install replicate express cors

const express = require('express');
const cors = require('cors');
const Replicate = require('replicate');

const app = express();
app.use(cors());
app.use(express.json());

const replicate = new Replicate({
    auth: process.env.REPLICATE_API_TOKEN, // Dapatkan kunci API di replicate.com
});

app.post('/api/ai-upscale', async (req, res) => {
    const { videoUrl, scale = 4, faceEnhance = true } = req.body;

    try {
        // Jalankan model AI Real-ESRGAN untuk pemulihan & peningkatan resolusi 4K
        const output = await replicate.run(
            "nightmareai/real-esrgan:f121d640bd286e1fdc67f9799164c1d5be36ff74576ee11c803ae5b665dd46aa",
            {
                input: {
                    video: videoUrl,
                    scale: scale,
                    face_enhance: faceEnhance
                }
            }
        );

        res.json({
            status: 'success',
            upscaledUrl: output
        });
    } catch (error) {
        res.status(500).json({ error: 'Pemprosesan AI gagal', detail: error.message });
    }
});

app.listen(3001, () => console.log('Pelayan Real-ESRGAN AI berjalan di port 3001'));
<?php
// Simpan sebagai api.php di pelayan hosting anda (PHP 7.4+ dengan cURL)
header("Access-Control-Allow-Origin: *");
header("Access-Control-Allow-Methods: GET, POST, OPTIONS");
header("Access-Control-Allow-Headers: Content-Type");

if (isset($_GET['action']) && $_GET['action'] === 'download' && !empty($_GET['fileUrl'])) {
    $fileUrl = $_GET['fileUrl'];
    $filename = isset($_GET['filename']) ? basename($_GET['filename']) : 'TikSave_Video.mp4';

    header('Content-Description: File Transfer');
    header('Content-Type: application/octet-stream');
    header('Content-Disposition: attachment; filename="' . $filename . '"');

    $ch = curl_init($fileUrl);
    curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
    curl_setopt($ch, CURLOPT_USERAGENT, 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)');
    curl_exec($ch);
    curl_close($ch);
    exit;
}

header("Content-Type: application/json; charset=UTF-8");
if (empty($_GET['url'])) {
    http_response_code(400);
    echo json_encode(["code" => -1, "msg" => "Parameter URL TikTok diperlukan."]);
    exit;
}

$tiktokUrl = urlencode($_GET['url']);

curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);
curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
